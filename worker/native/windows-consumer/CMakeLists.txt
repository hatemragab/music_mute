set(MUSICMUTE_PACKAGE_AUTHORITY "store" CACHE STRING "Compiled package authority: store or development-test")
set_property(CACHE MUSICMUTE_PACKAGE_AUTHORITY PROPERTY STRINGS store development-test)
set(MUSICMUTE_DEVELOPMENT_CERT_SHA256 "" CACHE STRING "Exact development signing certificate DER SHA256")
if(MUSICMUTE_PACKAGE_AUTHORITY STREQUAL "development-test")
  string(LENGTH "${MUSICMUTE_DEVELOPMENT_CERT_SHA256}" development_certificate_length)
  if(NOT development_certificate_length EQUAL 64 OR NOT MUSICMUTE_DEVELOPMENT_CERT_SHA256 MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "development-test requires MUSICMUTE_DEVELOPMENT_CERT_SHA256 as 64 lowercase hexadecimal characters")
  endif()
  configure_file(development_certificate_template.hpp musicmute_development_certificate.generated.hpp @ONLY)
  add_compile_definitions(MUSICMUTE_DEVELOPMENT_TEST_PACKAGE=1)
  include_directories("${CMAKE_CURRENT_BINARY_DIR}")
  link_libraries(crypt32)
elseif(NOT MUSICMUTE_PACKAGE_AUTHORITY STREQUAL "store")
  message(FATAL_ERROR "MUSICMUTE_PACKAGE_AUTHORITY must be store or development-test")
elseif(NOT MUSICMUTE_DEVELOPMENT_CERT_SHA256 STREQUAL "")
  message(FATAL_ERROR "Store authority cannot carry a development certificate pin")
endif()
